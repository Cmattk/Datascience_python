----: Create and Manage Group Roles ----

-- 'marketing' group role:
corp=# CREATE ROLE marketing NOINHERIT;
CREATE ROLE

-- 'planning' group role:
corp=# CREATE ROLE planning NOINHERIT;
CREATE ROLE

-- Grant all privileges on 'contacts' table to 'marketing':
corp=# GRANT ALL ON contacts TO marketing;
GRANT

-- Grant all privileges on 'forecasts' table to 'planning':
corp=# GRANT ALL ON forecasts TO planning;
GRANT

-- Add 'dave' as a member of 'marketing':
corp=# GRANT marketing TO dave;
GRANT ROLE

-- Add 'planning' as a member of 'marketing':
GRANT marketing TO planning;
GRANT ROLE

-- Now, the role dave can select data from the forecasts table:
corp=# SELECT * FROM forecasts;
